// B-3 훅 규칙 위반 채증 (내부 검증용)
import { useRef, useState } from 'react';

// 1. 조건문 안에서 훅을 부른다
export function ConditionalToggle({ expandable }: { expandable: boolean }) {
  if (expandable) {
    const [open, setOpen] = useState(false);
    return <button onClick={() => setOpen(!open)}>{open ? '접기' : '더보기'}</button>;
  }

  return null;
}

// 2. 반복문 안에서 훅을 부른다
export function LoopToggle({ captions }: { captions: string[] }) {
  const shown: string[] = [];

  for (const caption of captions) {
    const [open] = useState(false);
    shown.push(open ? caption : caption.slice(0, 10));
  }

  return <p>{shown.join(' / ')}</p>;
}

// 3. 이름이 use 로 시작하지 않는 함수에서 훅을 부른다
export function makeToggle(initialOn: boolean) {
  const [on, setOn] = useState(initialOn);

  return { on, toggle: () => setOn(!on) };
}

// 4. 커스텀 훅 안에서 조건부로 다른 훅을 부른다 (훅 안이라도 규칙은 똑같다)
export function useCommentDraft(autoFocus: boolean) {
  const [content, setContent] = useState('');

  if (autoFocus) {
    const inputRef = useRef<HTMLInputElement>(null);
    return { content, setContent, inputRef };
  }

  return { content, setContent, inputRef: null };
}

// 5. 이른 return 뒤에서 훅을 부른다 (조건문이 안 보여도 조건부 호출이다)
export function EarlyReturnCard({ content }: { content: string | null }) {
  if (content === null) {
    return null;
  }

  const [open, setOpen] = useState(false);

  return <button onClick={() => setOpen(!open)}>{open ? content : '더보기'}</button>;
}

// 6. 5 번과 같은데 이른 return 앞에 훅이 하나 먼저 있다
export function EarlyReturnAfterHook({ content }: { content: string | null }) {
  const [liked, setLiked] = useState(false);

  if (content === null) {
    return null;
  }

  const [open, setOpen] = useState(false);

  return (
    <button onClick={() => { setLiked(!liked); setOpen(!open); }}>
      {open ? content : '더보기'} {liked ? '♥' : '♡'}
    </button>
  );
}

· 위반 다섯 가지가 전부 react-hooks/rules-of-hooks 한 규칙에서 나온다.
  B-2 가 본 세 가지(조건·반복·비컴포넌트)에 커스텀 훅 안 위반이 그대로 붙는다.

· ⚠️ "Did you accidentally call a React Hook after an early return?" 꼬리말이
  붙는 자리가 직관과 반대다. 실제로 이른 return 뒤에 훅을 둔 5·6 번에는
