// B-2 훅 규칙 위반 채증 (내부 검증용)
import { useState } from 'react';

// 1. 조건문 안에서 훅을 부른다
export function ConditionalHook({ showLike }: { showLike: boolean }) {
  if (showLike) {
    const [liked, setLiked] = useState(false);
    return <button onClick={() => setLiked(!liked)}>{liked ? '♥' : '♡'}</button>;
  }

  return null;
}

// 2. 반복문 안에서 훅을 부른다
export function LoopHook({ names }: { names: string[] }) {
  const labels: string[] = [];

  for (const name of names) {
    const [liked] = useState(false);
    labels.push(liked ? `${name} ♥` : `${name} ♡`);
  }

  return <p>{labels.join(', ')}</p>;
}

// 3. 컴포넌트가 아닌 그냥 함수에서 훅을 부른다
export function makeLikeState() {
  const [liked, setLiked] = useState(false);
  return { liked, setLiked };
}
