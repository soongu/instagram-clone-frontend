────────────────── 소스 (가) if 블록 안으로 옮긴 판 ──────────────────

// PostBody 의 useToggle 호출을 if (isLong) 안으로 옮긴 모습
import { LikeButton } from '../src/components/LikeButton';
import { Button } from '../src/components/Button';
import { useToggle } from '../src/hooks/useToggle';

const CAPTION_LIMIT = 10;

interface PostBodyProps {
  username: string;
  content: string;
  liked: boolean;
  likeCount: number;
  commentCount: number;
  onToggle: () => void;
}

export function PostBody({
  username,
  content,
  liked,
  likeCount,
  commentCount,
  onToggle,
}: PostBodyProps) {
  const isLong = content.length > CAPTION_LIMIT;

  if (isLong) {
    const [captionOpen, toggleCaption] = useToggle();
    const shownContent = captionOpen
      ? content
      : `${content.slice(0, CAPTION_LIMIT).trimEnd()}...`;

    return (
      <>
        <LikeButton liked={liked} likeCount={likeCount} onToggle={onToggle} />
        <p className="post-content">
          <strong>{username}</strong> {shownContent}
          <Button className="caption-toggle" onClick={toggleCaption}>
            {captionOpen ? '접기' : '더 보기'}
          </Button>
        </p>
        <p className="post-comments">댓글 {commentCount}개 모두 보기</p>
      </>
    );
  }

  return (
    <>
      <LikeButton liked={liked} likeCount={likeCount} onToggle={onToggle} />
      <p className="post-content">
        <strong>{username}</strong> {content}
      </p>
      <p className="post-comments">댓글 {commentCount}개 모두 보기</p>
    </>
  );
}

─────────────────────── 확인한 것 ────────────────────────

· 두 판 모두 메시지가 같다. "Did you accidentally call a React Hook after an
  early return?" 꼬리말은 어느 쪽에도 안 붙는다. 그 꼬리말은 조건부 훅 앞에
  다른 훅이 이미 있는 모습(교안 Step 5 의 useCommentDraft)에서만 붙었다.

· 메시지에 불린 이름은 useState 가 아니라 useToggle 이다. 린트는 use 로
  시작하는 이름을 전부 훅으로 보고 잡는다.
